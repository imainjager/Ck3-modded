# A vassal gives the thief up (or the patron pays up if the thief is already in your dungeon).
ie_theft_vassal_success_effect = {
	if = {
		limit = { scope:thief = { is_imprisoned = yes } }
		add_gold = {
			value = ie_theft_loot_value
			multiply = 0.5
		}
		ie_theft_end_story_effect = yes
	}
	else = {
		ie_theft_set_method_effect = { METHOD = 12 }
		trigger_event = {
			id = ie_theft.0008
			days = 4
		}
	}
}

# A foreign ruler gives the thief up (or pays reparations if the thief is already in your dungeon).
ie_theft_foreign_success_effect = {
	if = {
		limit = { scope:thief = { is_imprisoned = yes } }
		add_gold = {
			value = ie_theft_loot_value
			multiply = 0.5
		}
		ie_theft_end_story_effect = yes
	}
	else = {
		ie_theft_set_method_effect = { METHOD = $METHOD$ }
		trigger_event = {
			id = ie_theft.0008
			days = 6
		}
	}
}

# Any foreign dispute that goes nowhere ends here.
ie_theft_foreign_failure_effect = {
	ie_theft_unrecovered_effect = yes
}
