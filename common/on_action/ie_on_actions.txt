# Hook the reworked events into the game's yearly random pulse for playable characters.
# on_actions with the same name are additive, so vanilla events keep firing.
random_yearly_playable_pulse = {
	on_actions = { ie_yearly_flavor_pulse }
}

ie_yearly_flavor_pulse = {
	trigger = {
		is_ai = no
	}
	random_events = {
		chance_to_happen = 30
		100 = ie_theft.0001
		100 = 0
	}
}
