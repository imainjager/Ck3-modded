# War fought because a foreign ruler shelters (or sponsored) a thief and refuses justice.
# Only started from the Theft at Court story, never from the declare-war screen.
ie_stolen_justice_cb = {
	group = event
	icon = invasion

	allowed_for_character = {}
	allowed_against_character = {}

	cost = {}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = ie_stolen_justice_cb_victory_desc
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_victory_effect = yes

		scope:attacker = {
			add_prestige = 150
			add_gold = ie_theft_loot_value
		}
		scope:defender = {
			add_prestige = -150
		}

		# The thief is handed over
		scope:attacker = {
			trigger_event = {
				id = ie_theft.0008
				days = 3
			}
		}

		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = ie_stolen_justice_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = ie_stolen_justice_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_prestige = -100
		}
		add_legitimacy_attacker_defeat_effect = yes
		add_truce_attacker_defeat_effect = yes
	}

	on_invalidated = {
	}

	on_invalidated_desc = {
		desc = ie_stolen_justice_cb_invalidated_desc
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	transfer_behavior = transfer

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "IE_STOLEN_JUSTICE_WAR_NAME"
	cb_name_no_target = "IE_STOLEN_JUSTICE_WAR_NAME"

	interface_priority = 0

	attacker_wargoal_percentage = 0.8

	should_show_war_goal_subview = no
}
