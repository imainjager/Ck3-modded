namespace = ie_theft

##################################################################
# THEFT AT COURT - part 2: the thief at large, and the fallout
#
# 0006 The Thief Surfaces       - route 1: sheltered at a foreign court
# 0007 A Matter of Honor        - that ruler refuses; diplomacy, intrigue or war
# 0030 The Protected Thief      - route 2: a vassal shelters (or paid for) the thief
# 0040 The Bandit Chief         - route 3: the thief leads brigands in your realm
# 0008 Justice Delivered        - the thief is in your hands (any route)
# 0009 Stolen Gold, Spent Well  - the unrecovered gold causes trouble at home
# 0010 Old Habits               - a pardoned thief steals again
# 0011 A Blood Debt             - a relative seeks revenge
# 0090 The Court Remembers      - epilogue: how this changed your standing
#
# Capture method (story var "method"):
# 6 abduction, 7 extradition, 8 war, 9 bounty, 10 hunted down,
# 11 long pursuit, 12 vassal surrendered, 13 hostage leverage, 14 intimidation
##################################################################


#####################################
# 0006 The Thief Surfaces
#####################################
ie_theft.0006 = {
	type = character_event
	title = ie_theft.0006.t
	desc = ie_theft.0006.desc
	theme = diplomacy

	trigger = {
		any_owned_story = {
			type = ie_story_theft
			exists = var:thief_at_large
			var:thief = { is_alive = yes }
		}
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
		ie_theft_pick_foreign_effect = yes
		if = {
			limit = { exists = scope:harbor_ruler }
			scope:harbor_ruler = { add_courtier = scope:thief }
		}
	}

	# Covert action: have him taken or silenced
	option = {
		name = ie_theft.0006.a
		trigger = {
			OR = {
				exists = scope:spymaster
				intrigue >= 12
				has_trait = schemer
				has_perk = digging_for_dirt_perk
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
			just = miniscule_stress_impact_gain
			compassionate = miniscule_stress_impact_gain
			deceitful = miniscule_stress_impact_loss
		}
		random_list = {
			35 = {
				desc = ie_theft.out.captured
				modifier = {
					add = 15
					exists = scope:spymaster
					scope:spymaster = { intrigue >= 12 }
				}
				modifier = {
					add = 15
					intrigue >= 16
				}
				modifier = {
					add = 10
					has_perk = kidnapper_perk
				}
				ie_theft_set_method_effect = { METHOD = 6 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			20 = {
				desc = ie_theft.out.thief_dead
				add_gold = {
					value = ie_theft_loot_value
					multiply = 0.3
				}
				ie_theft_rep_effect = {
					VAR = ie_fear
					AMOUNT = 1
				}
				scope:thief = {
					death = {
						death_reason = death_murder
						killer = root
					}
				}
				ie_theft_end_story_effect = yes
			}
			35 = {
				desc = ie_theft.out.incident
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -30
					}
				}
				add_prestige = -25
				ie_theft_unrecovered_effect = yes
			}
		}
	}


	# Negotiate a deal
	option = {
		name = ie_theft.0006.c
		remove_short_term_gold = { value = ie_theft_loot_value multiply = 0.2 }
		stress_impact = {
			shy = miniscule_stress_impact_gain
			gregarious = miniscule_stress_impact_loss
		}
		random_list = {
			35 = {
				desc = ie_theft.out.surrender_deal
				modifier = {
					add = 10
					diplomacy >= 10
				}
				modifier = {
					add = 15
					diplomacy >= 16
				}
				modifier = {
					add = 10
					has_perk = friendly_counsel_perk
				}
				modifier = {
					add = 25
					scope:harbor_ruler = {
						opinion = {
							target = root
							value >= 20
						}
					}
				}
				modifier = {
					add = 10
					has_trait = gregarious
				}
				scope:harbor_ruler = {
					add_hook = {
						target = root
						type = favor_hook
					}
				}
				ie_theft_rep_effect = {
					VAR = ie_loyalty
					AMOUNT = 1
				}
				ie_theft_set_method_effect = { METHOD = 7 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			50 = {
				desc = ie_theft.out.refused
				trigger_event = {
					id = ie_theft.0007
					days = 3
				}
			}
		}
	}

	# Intimidate them into surrendering him
	option = {
		name = ie_theft.0006.d
		trigger = {
			OR = {
				martial >= 12
				has_trait = brave
				has_trait = wrathful
				has_perk = dreadful_perk
			}
		}
		stress_impact = {
			craven = miniscule_stress_impact_gain
			brave = miniscule_stress_impact_loss
			compassionate = miniscule_stress_impact_gain
		}
		random_list = {
			35 = {
				desc = ie_theft.out.surrender_fear
				modifier = {
					add = 15
					martial >= 16
				}
				modifier = {
					add = 15
					has_perk = dreadful_perk
				}
				modifier = {
					add = 20
					scope:harbor_ruler = { highest_held_title_tier < root.highest_held_title_tier }
				}
				ie_theft_rep_effect = {
					VAR = ie_fear
					AMOUNT = 1
				}
				ie_theft_set_method_effect = { METHOD = 14 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			45 = {
				desc = ie_theft.out.refused
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -10
					}
				}
				trigger_event = {
					id = ie_theft.0007
					days = 3
				}
			}
		}
	}



	# Offer a bounty
	option = {
		name = ie_theft.0006.g
		remove_short_term_gold = { value = ie_theft_loot_value multiply = 0.4 }
		random_list = {
			55 = {
				desc = ie_theft.out.bounty_hunter
				ie_theft_set_method_effect = { METHOD = 9 }
				trigger_event = {
					id = ie_theft.0008
					months = { 3 8 }
				}
			}
			45 = {
				desc = ie_theft.out.nobody_takes
				ie_theft_unrecovered_effect = yes
			}
		}
	}

	# Swear an oath of vengeance
	option = {
		name = ie_theft.0006.h
		trigger = { has_trait = vengeful }
		ie_theft_make_rival_effect = yes
		stress_impact = {
			vengeful = minor_stress_impact_loss
		}
		random_list = {
			55 = {
				desc = ie_theft.out.long_pursuit
				ie_theft_set_method_effect = { METHOD = 11 }
				trigger_event = {
					id = ie_theft.0008
					years = { 1 3 }
				}
			}
			45 = {
				desc = ie_theft.out.nobody_takes
				ie_theft_unrecovered_effect = yes
			}
		}
	}

	# Let him go
	option = {
		name = ie_theft.0006.i
		stress_impact = {
			vengeful = miniscule_stress_impact_gain
			wrathful = miniscule_stress_impact_gain
			forgiving = miniscule_stress_impact_loss
		}
		ie_theft_unrecovered_effect = yes
	}
}


#####################################
# 0007 A Matter of Honor
#####################################
ie_theft.0007 = {
	type = character_event
	title = ie_theft.0007.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:story = { var:thief_sponsored ?= 1 } }
				desc = ie_theft.0007.desc.sponsor
			}
			desc = ie_theft.0007.desc.harbor
		}
	}
	theme = diplomacy
	left_portrait = {
		character = scope:harbor_ruler
		animation = scheme
	}

	trigger = {
		any_owned_story = { type = ie_story_theft }
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
	}

	# War
	option = {
		name = ie_theft.0007.a
		trigger = {
			NOT = { has_truce = scope:harbor_ruler }
			is_at_war = no
		}
		show_as_unavailable = { always = yes }
		ie_theft_set_method_effect = { METHOD = 8 }
		start_war = {
			cb = ie_stolen_justice_cb
			target = scope:harbor_ruler
		}
	}

	# Call in a hook
	option = {
		name = ie_theft.0007.b
		trigger = { has_hook = scope:harbor_ruler }
		use_hook = scope:harbor_ruler
		ie_theft_foreign_success_effect = { METHOD = 7 }
	}

	# Rally your allies to pressure them
	option = {
		name = ie_theft.0007.c
		trigger = {
			any_ally = { count >= 1 }
		}
		random_list = {
			40 = {
				desc = ie_theft.out.allies_prevail
				modifier = {
					add = 20
					any_ally = { count >= 2 }
				}
				modifier = {
					add = 10
					any_ally = { count >= 3 }
				}
				ie_theft_foreign_success_effect = { METHOD = 7 }
			}
			40 = {
				desc = ie_theft.out.allies_fail
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -10
					}
				}
				ie_theft_foreign_failure_effect = yes
			}
		}
	}

	# Pay their price
	option = {
		name = ie_theft.0007.d
		trigger = {
			NOT = { scope:story = { var:thief_sponsored ?= 1 } }
		}
		remove_short_term_gold = { value = ie_theft_loot_value multiply = 0.5 }
		ie_theft_rep_effect = {
			VAR = ie_mockery
			AMOUNT = 1
		}
		stress_impact = {
			arrogant = minor_stress_impact_gain
			greedy = minor_stress_impact_gain
		}
		random_list = {
			75 = {
				desc = ie_theft.out.surrender_deal
				ie_theft_set_method_effect = { METHOD = 7 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			15 = {
				desc = ie_theft.out.took_money
				ie_theft_unrecovered_effect = yes
			}
		}
	}


	# Demand reparations (the thief is in your dungeon)
	option = {
		name = ie_theft.0007.h
		trigger = {
			scope:story = { var:thief_sponsored ?= 1 }
			OR = {
				diplomacy >= 10
				has_trait = gregarious
			}
		}
		show_as_unavailable = {
			scope:story = { var:thief_sponsored ?= 1 }
		}
		random_list = {
			50 = {
				desc = ie_theft.out.reparations
				modifier = {
					add = 15
					diplomacy >= 14
				}
				add_gold = {
					value = ie_theft_loot_value
					multiply = 0.75
				}
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -15
					}
				}
				ie_theft_rep_effect = {
					VAR = ie_fear
					AMOUNT = 1
				}
				ie_theft_end_story_effect = yes
			}
			35 = {
				desc = ie_theft.out.refused
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -20
					}
				}
				ie_theft_end_story_effect = yes
			}
		}
	}

	# Expose them to every court (the thief is in your dungeon)
	option = {
		name = ie_theft.0007.i
		trigger = {
			scope:story = { var:thief_sponsored ?= 1 }
		}
		add_prestige = 100
		scope:harbor_ruler = {
			add_prestige = -100
			add_opinion = {
				target = root
				modifier = insulted_opinion
				opinion = -50
			}
		}
		ie_theft_rep_effect = {
			VAR = ie_fear
			AMOUNT = 1
		}
		stress_impact = {
			honest = miniscule_stress_impact_loss
			just = miniscule_stress_impact_loss
		}
		ie_theft_end_story_effect = yes
	}


	# Swallow your pride
	option = {
		name = ie_theft.0007.g
		add_prestige = -75
		ie_theft_rep_effect = {
			VAR = ie_mockery
			AMOUNT = 1
		}
		stress_impact = {
			arrogant = minor_stress_impact_gain
			vengeful = minor_stress_impact_gain
			wrathful = minor_stress_impact_gain
			humble = miniscule_stress_impact_loss
			craven = miniscule_stress_impact_loss
		}
		if = {
			limit = {
				scope:story = { var:thief_sponsored ?= 1 }
			}
			ie_theft_end_story_effect = yes
		}
		else = {
			ie_theft_unrecovered_effect = yes
		}
	}
}

	}

		any_owned_story = {

						}

					add = 15
					diplomacy >= 12
				}
				modifier = {
					add = -25

					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -20
					}
				}
				ie_theft_unrecovered_effect = yes

				has_perk = dreadful_perk
			}
		}
		stress_impact = {
			humble = minor_stress_impact_gain
			compassionate = miniscule_stress_impact_gain
			arrogant = miniscule_stress_impact_loss
		}
		random_list = {
			50 = {
				desc = ie_theft.out.surrender_fear
				modifier = {
					add = 15
					martial >= 16
				}
				modifier = {
					add = 15
					has_perk = dreadful_perk
				}
				scope:harbor_vassal = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -30
					}
				}
				ie_theft_rep_effect = {
					VAR = ie_fear
					AMOUNT = 1
				}
				ie_theft_vassal_success_effect = yes
			}
			40 = {
				desc = ie_theft.out.vassal_defiant
				scope:harbor_vassal = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -50
					}
				}
				ie_theft_rep_effect = {
					VAR = ie_mockery
					AMOUNT = 1
				}
				ie_theft_unrecovered_effect = yes
			}
		}
	}

	# Strike a bargain
	option = {
		name = ie_theft.0030.d

				diplomacy >= 10

				has_perk = diplomat_perk

		}

	# Blackmail them
	option = {
		name = ie_theft.0030.e

	# Imprison the vassal for harbouring a thief

		name = ie_theft.0030.f

			honest = minor_stress_impact_gain

	# Let it lie
	option = {
		name = ie_theft.0030.g

			add_opinion = {
				target = root
				modifier = grateful_opinion

			}
		}
		ie_theft_rep_effect = {
			VAR = ie_mockery
			AMOUNT = 1
		}
		stress_impact = {
			vengeful = miniscule_stress_impact_gain
			wrathful = miniscule_stress_impact_gain
		}
		ie_theft_unrecovered_effect = yes
	}
}


#####################################
# 0040 The Bandit Chief
#####################################
ie_theft.0040 = {
	type = character_event
	title = ie_theft.0040.t
	desc = ie_theft.0040.desc
	theme = war

	trigger = {
		any_owned_story = {
			type = ie_story_theft
			var:thief = { is_alive = yes }
		}
		any_realm_county = { count >= 1 }
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
		random_realm_county = {
			save_scope_as = haven_county
		}
		scope:haven_county = {
			add_county_modifier = {
				modifier = ie_bandit_haven_modifier
				years = 6
			}
		}
	}

	# Hunt him down (you lead if you can fight; otherwise the marshal does)
	option = {
		name = ie_theft.0040.a
		stress_impact = {
			craven = minor_stress_impact_gain
			brave = miniscule_stress_impact_loss
		}
		random_list = {
			40 = {
				desc = ie_theft.out.captured
				modifier = {
					add = 10
					martial >= 10
				}
				modifier = {
					add = 15
					martial >= 16
				}
				modifier = {
					add = 10
					has_trait = brave
				}
				modifier = {
					add = 10
					has_perk = hit_and_run_perk
				}
				modifier = {
					add = 15
					exists = scope:marshal
					scope:marshal = { martial >= 12 }
				}
				scope:haven_county = {
					remove_county_modifier = ie_bandit_haven_modifier
				}
				add_prestige = 50
				ie_theft_set_method_effect = { METHOD = 10 }
				trigger_event = {
					id = ie_theft.0008
					days = 10
				}
			}
