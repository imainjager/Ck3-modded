namespace = ie_theft

##################################################################
# THEFT AT COURT - part 1: the investigation
#
# 0001 A Hole in the Treasury   - entry; choose how to investigate
# 0020 The Interrogation        - when you question the court yourself
# 0013 The Confession           - the thief comes forward (amnesty / conscience)
# 0002 The Culprit Revealed     - thief identified; choose their fate
# 0003 A Convenient Culprit     - investigation points at the wrong person
# 0004 A Second Look            - keep digging after a wrong lead
# 0005 The Truth Comes Out      - the innocent you punished is cleared
#
# Story variables used here:
#   motive        1 greed, 2 desperation, 3 paid by a vassal, 4 paid by a foreign ruler
#   investigator  who investigated (see ie_theft_set_investigator_effect)
#   approach      how you questioned the court yourself
##################################################################


#####################################
# 0001 A Hole in the Treasury
#####################################
ie_theft.0001 = {
	type = character_event
	title = ie_theft.0001.t
	desc = ie_theft.0001.desc
	theme = stewardship
	cooldown = { years = 6 }

	trigger = {
		is_ai = no
		is_available_adult = yes
		NOT = { has_character_flag = ie_global_cooldown }
		NOT = { has_character_flag = ie_theft_cooldown }
		NOT = { any_owned_story = { type = ie_story_theft } }
		gold >= ie_theft_loot_value
		any_courtier = {
			count >= 2
			ie_theft_candidate_trigger = yes
		}
	}

	immediate = {
		add_character_flag = {
			flag = ie_global_cooldown
			months = 2
		}
		# Greedy or deceitful courtiers are likelier suspects
		if = {
			limit = {
				any_courtier = {
					ie_theft_candidate_trigger = yes
					OR = {
						has_trait = greedy
						has_trait = deceitful
					}
				}
			}
			random_courtier = {
				limit = {
					ie_theft_candidate_trigger = yes
					OR = {
						has_trait = greedy
						has_trait = deceitful
					}
				}
				save_scope_as = thief
			}
		}
		else = {
			random_courtier = {
				limit = { ie_theft_candidate_trigger = yes }
				save_scope_as = thief
			}
		}
		random_courtier = {
			limit = {
				ie_theft_candidate_trigger = yes
				NOT = { this = scope:thief }
			}
			save_scope_as = scapegoat
		}
		add_gold = {
			value = ie_theft_loot_value
			multiply = -1
		}
		create_story = {
			type = ie_story_theft
			save_scope_as = story
		}
		scope:story = {
			set_variable = {
				name = thief
				value = scope:thief
			}
			set_variable = {
				name = scapegoat
				value = scope:scapegoat
			}
			set_variable = {
				name = investigator
				value = 0
			}
		}
		# Why did they do it?
		random_list = {
			40 = {
				modifier = {
					add = 30
					scope:thief = { has_trait = greedy }
				}
				scope:story = {
					set_variable = {
						name = motive
						value = 1
					}
				}
			}
			25 = {
				modifier = {
					add = 15
					scope:thief = { has_trait = humble }
				}
				scope:story = {
					set_variable = {
						name = motive
						value = 2
					}
				}
			}
			15 = {
				trigger = {
					any_vassal = { highest_held_title_tier >= tier_county }
				}
				scope:story = {
					set_variable = {
						name = motive
						value = 3
					}
				}
			}
			20 = {
				trigger = {
					any_neighboring_and_across_water_top_liege_realm_owner = { is_alive = yes }
				}
				scope:story = {
					set_variable = {
						name = motive
						value = 4
					}
				}
			}
		}
		ie_theft_load_scopes_effect = yes
	}

	# Question the court yourself - how is up to you
	option = {
		name = ie_theft.0001.a
		custom_tooltip = ie_theft.0001.a.tt
		trigger_event = {
			id = ie_theft.0020
			days = 1
		}
	}

	# Steward audits the books
	option = {
		name = ie_theft.0001.b
		trigger = { exists = scope:steward }
		ie_theft_set_investigator_effect = { INV = 2 }
		stress_impact = {
			honest = miniscule_stress_impact_loss
		}
		random_list = {
			40 = {
				desc = ie_theft.out.identified
				modifier = {
					add = 15
					scope:steward = { stewardship >= 10 }
				}
				modifier = {
					add = 20
					scope:steward = { stewardship >= 16 }
				}
				modifier = {
					add = 10
					scope:steward = { has_perk = detailed_ledgers_perk }
				}
				modifier = {
					add = -15
					scope:thief = { has_trait = deceitful }
				}
				trigger_event = {
					id = ie_theft.0002
					days = 3
				}
			}
			45 = {
				desc = ie_theft.out.wrong_lead
				trigger_event = {
					id = ie_theft.0003
					days = 3
				}
			}
		}
	}

	# Spymaster uses informants
	option = {
		name = ie_theft.0001.c
		trigger = { exists = scope:spymaster }
		ie_theft_set_investigator_effect = { INV = 3 }
		stress_impact = {
			honest = miniscule_stress_impact_gain
			just = miniscule_stress_impact_gain
		}
		random_list = {
			40 = {
				desc = ie_theft.out.identified
				modifier = {
					add = 15
					scope:spymaster = { intrigue >= 10 }
				}
				modifier = {
					add = 20
					scope:spymaster = { intrigue >= 16 }
				}
				modifier = {
					add = 10
					scope:spymaster = { has_perk = digging_for_dirt_perk }
				}
				modifier = {
					add = -10
					scope:thief = { has_trait = paranoid }
				}
				trigger_event = {
					id = ie_theft.0002
					days = 5
				}
			}
			45 = {
				desc = ie_theft.out.wrong_lead
				trigger_event = {
					id = ie_theft.0003
					days = 5
				}
			}
		}
	}

	# Chaplain hears confessions and swears the court on relics
	option = {
		name = ie_theft.0001.d
		trigger = { exists = scope:chaplain }
		ie_theft_set_investigator_effect = { INV = 5 }
		random_list = {
			40 = {
				desc = ie_theft.out.identified
				modifier = {
					add = 15
					scope:chaplain = { learning >= 10 }
				}
				modifier = {
					add = 10
					scope:chaplain = { has_trait = zealous }
				}
				modifier = {
					add = -20
					scope:thief = { has_trait = zealous }
				}
				modifier = {
					add = -10
					scope:thief = { has_trait = deceitful }
				}
				trigger_event = {
					id = ie_theft.0002
					days = 4
				}
			}
			45 = {
				desc = ie_theft.out.wrong_lead
				trigger_event = {
					id = ie_theft.0003
					days = 4
				}
			}
		}
	}

	# Paranoid: trust nobody and lock the court down
	option = {
		name = ie_theft.0001.e
		trigger = { has_trait = paranoid }
		ie_theft_set_investigator_effect = { INV = 8 }
		ie_theft_court_mood_effect = { OPINION = -10 }
		stress_impact = {
			paranoid = minor_stress_impact_loss
		}
		random_list = {
			55 = {
				desc = ie_theft.out.identified
				modifier = {
					add = 10
					has_trait = shrewd
				}
				trigger_event = {
					id = ie_theft.0002
					days = 2
				}
			}
			30 = {
				desc = ie_theft.out.wrong_lead
				trigger_event = {
					id = ie_theft.0003
					days = 2
				}
			}
		}
	}

	# Cruel: take the likeliest suspect to the cellars
	option = {
		name = ie_theft.0001.f
		trigger = {
			OR = {
				has_trait = sadistic
				has_perk = torturer_perk
				has_trait = torturer
			}
		}
		ie_theft_set_investigator_effect = { INV = 9 }
		add_dread = minor_dread_gain
		ie_theft_rep_effect = {
			VAR = ie_fear
			AMOUNT = 1
		}
		stress_impact = {
			sadistic = minor_stress_impact_loss
			compassionate = minor_stress_impact_gain
			forgiving = miniscule_stress_impact_gain
			just = miniscule_stress_impact_gain
		}
		random_list = {
			50 = {
				desc = ie_theft.out.confession_true
				trigger_event = {
					id = ie_theft.0002
					days = 2
				}
			}
			50 = {
				desc = ie_theft.out.false_confession
				scope:story = {
					set_variable = {
						name = scapegoat_tortured
						value = 1
					}
				}
				trigger_event = {
					id = ie_theft.0003
					days = 2
				}
			}
		}
	}

	# Open-handed: offer a reward for information
	option = {
		name = ie_theft.0001.g
		trigger = {
			OR = {
				has_trait = generous
				has_trait = gregarious
				has_trait = trusting
			}
		}
		ie_theft_set_investigator_effect = { INV = 6 }
		add_gold = {
			value = ie_theft_loot_value
			multiply = -0.15
		}
		stress_impact = {
			greedy = minor_stress_impact_gain
			generous = miniscule_stress_impact_loss
		}
		random_list = {
			45 = {
				desc = ie_theft.out.tip_true
				trigger_event = {
					id = ie_theft.0002
					days = 3
				}
			}
			40 = {
				desc = ie_theft.out.tip_false
				trigger_event = {
					id = ie_theft.0003
					days = 3
				}
			}
		}
	}

	# Merciful: offer amnesty to whoever returns the gold
	option = {
		name = ie_theft.0001.h
		trigger = {
			OR = {
				has_trait = compassionate
				has_trait = forgiving
				has_trait = just
			}
		}
		ie_theft_set_investigator_effect = { INV = 7 }
		stress_impact = {
			compassionate = minor_stress_impact_loss
			forgiving = minor_stress_impact_loss
			vengeful = miniscule_stress_impact_gain
			wrathful = miniscule_stress_impact_gain
		}
		random_list = {
			35 = {
				desc = ie_theft.out.thief_confesses
				trigger_event = {
					id = ie_theft.0013
					days = 2
				}
			}
			55 = {
				desc = ie_theft.out.nobody_comes
				ie_theft_rep_effect = {
					VAR = ie_mockery
					AMOUNT = 1
				}
				trigger_event = {
					id = ie_theft.0004
					months = 1
				}
			}
		}
	}

	# Delegation: let the steward handle it quietly
	option = {
		name = ie_theft.0001.i
		trigger = { exists = scope:steward }
		stress_impact = {
			lazy = miniscule_stress_impact_loss
			diligent = miniscule_stress_impact_gain
		}
		random_list = {
			45 = {
				desc = ie_theft.out.steward_resolves
				modifier = {
					add = 15
					scope:steward = { stewardship >= 10 }
				}
				modifier = {
					add = 15
					scope:steward = { stewardship >= 16 }
				}
				send_interface_toast = {
					type = event_toast_text_and_effect_good
					title = ie_theft.0001.i.success.t
					desc = ie_theft.0001.i.success.desc
					left_icon = scope:steward
					right_icon = scope:thief
					add_gold = {
						value = ie_theft_loot_value
						multiply = 0.5
					}
					scope:thief = {
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -15
						}
					}
					ie_theft_end_story_effect = yes
				}
			}
			55 = {
				desc = ie_theft.out.thief_escapes
				send_interface_toast = {
					type = event_toast_text_and_effect_bad
					title = ie_theft.0001.i.fail.t
					desc = ie_theft.0001.i.fail.desc
					left_icon = scope:steward
					ie_theft_thief_escapes_effect = yes
				}
			}
		}
	}

	# Accept the loss and move on
	option = {
		name = ie_theft.0001.j
		add_character_modifier = {
			modifier = ie_lax_treasury_modifier
			years = 3
		}
		ie_theft_rep_effect = {
			VAR = ie_mockery
			AMOUNT = 1
		}
		stress_impact = {
			just = miniscule_stress_impact_gain
			diligent = miniscule_stress_impact_gain
			wrathful = miniscule_stress_impact_gain
			lazy = miniscule_stress_impact_loss
			content = miniscule_stress_impact_loss
		}
		ie_theft_end_story_effect = yes
	}
}

	}

	option = {
		name = ie_theft.0004.d
		ie_theft_thief_escapes_effect = yes
	}

	type = character_event
	title = ie_theft.0005.t
	desc = {
		desc = ie_theft.0005.sum
		triggered_desc = {

		}

	}

	left_portrait = {
		character = scope:scapegoat

	}

		any_owned_story = {
			type = ie_story_theft
			exists = var:scapegoat_punished
		}
	}

		ie_theft_load_scopes_effect = yes
	}

	option = {
		name = ie_theft.0005.a

			limit = {
				NOT = { scope:scapegoat = { is_courtier_of = root } }
			}
			add_courtier = scope:scapegoat
		}

			value = ie_theft_loot_value
			multiply = -0.5
		}
		add_prestige = -50
		ie_theft_rep_effect = {
			VAR = ie_loyalty
			AMOUNT = 2
		}
		stress_impact = {
			arrogant = minor_stress_impact_gain
			honest = miniscule_stress_impact_loss
			just = miniscule_stress_impact_loss
		}
	}

	# Release them quietly
	option = {
		name = ie_theft.0005.b
		scope:scapegoat = {
			if = {
				limit = { is_imprisoned = yes }
				release_from_prison = yes
			}

				target = root

			}

			AMOUNT = 1
		}

			add_opinion = {
				target = root

			if = {
				limit = { is_imprisoned = yes }
				release_from_prison = yes
			}

				target = root
				modifier = grateful_opinion
				opinion = 10
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
			just = miniscule_stress_impact_gain
			deceitful = miniscule_stress_impact_loss
		}
	}

	# Double down
	option = {
		name = ie_theft.0005.d
		add_dread = minor_dread_gain
		ie_theft_rep_effect = {
			VAR = ie_fear
			AMOUNT = 1
		}

		stress_impact = {
			honest = minor_stress_impact_gain
			just = minor_stress_impact_gain
			arrogant = miniscule_stress_impact_loss
		}
	}
