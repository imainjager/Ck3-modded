namespace = ie_theft

##################################################################
# THEFT AT COURT - a branching story (story cycle: ie_story_theft)
#
# 0001 A Hole in the Treasury   - entry; choose how to investigate
# 0002 The Culprit Revealed     - thief identified; choose their fate
# 0003 A Convenient Culprit     - investigation points at the wrong person
# 0004 A Second Look            - keep digging after a wrong lead
# 0005 The Truth Comes Out      - the innocent you punished is cleared
# 0006 The Thief Surfaces       - thief turns up at a foreign court
# 0007 A Matter of Honor        - that ruler refuses; diplomacy or war
# 0008 Justice Delivered        - thief in your hands (any route)
# 0009 Stolen Gold, Spent Well  - the loot funds bandits in your realm
# 0010 Old Habits               - a pardoned thief steals again
# 0011 A Blood Debt             - a relative seeks revenge
#
# How the thief was identified is stored in story var "method":
# 1 you, 2 steward, 3 spymaster, 4 private talk, 5 second look,
# 6 abduction, 7 extradition, 8 war, 9 bounty
##################################################################


#####################################
# 0001 A Hole in the Treasury
#####################################
ie_theft.0001 = {
	type = character_event
	title = ie_theft.0001.t
	desc = ie_theft.0001.desc
	theme = stewardship
	cooldown = { years = 6 }

	trigger = {
		is_ai = no
		is_available_adult = yes
		NOT = { has_character_flag = ie_global_cooldown }
		NOT = { has_character_flag = ie_theft_cooldown }
		NOT = { any_owned_story = { type = ie_story_theft } }
		gold >= ie_theft_loot_value
		any_courtier = {
			count >= 2
			ie_theft_candidate_trigger = yes
		}
	}

	immediate = {
		add_character_flag = {
			flag = ie_global_cooldown
			months = 2
		}
		# Greedy or deceitful courtiers are likelier suspects
		if = {
			limit = {
				any_courtier = {
					ie_theft_candidate_trigger = yes
					OR = {
						has_trait = greedy
						has_trait = deceitful
					}
				}
			}
			random_courtier = {
				limit = {
					ie_theft_candidate_trigger = yes
					OR = {
						has_trait = greedy
						has_trait = deceitful
					}
				}
				save_scope_as = thief
			}
		}
		else = {
			random_courtier = {
				limit = { ie_theft_candidate_trigger = yes }
				save_scope_as = thief
			}
		}
		random_courtier = {
			limit = {
				ie_theft_candidate_trigger = yes
				NOT = { this = scope:thief }
			}
			save_scope_as = scapegoat
		}
		add_gold = {
			value = ie_theft_loot_value
			multiply = -1
		}
		create_story = {
			type = ie_story_theft
			save_scope_as = story
		}
		scope:story = {
			set_variable = {
				name = thief
				value = scope:thief
			}
			set_variable = {
				name = scapegoat
				value = scope:scapegoat
			}
		}
		if = {
			limit = { exists = cp:councillor_steward }
			cp:councillor_steward = { save_scope_as = steward }
		}
		if = {
			limit = { exists = cp:councillor_spymaster }
			cp:councillor_spymaster = { save_scope_as = spymaster }
		}
	}

	# Question the court yourself
	option = {
		name = ie_theft.0001.a
		stress_impact = {
			compassionate = minor_stress_impact_gain
		}
		ie_theft_set_method_effect = { METHOD = 1 }
		random_list = {
			40 = {
				modifier = {
					add = 15
					intrigue >= 10
				}
				modifier = {
					add = 15
					intrigue >= 16
				}
				modifier = {
					add = 10
					has_trait = shrewd
				}
				trigger_event = ie_theft.0002
			}
			45 = {
				trigger_event = ie_theft.0003
			}
		}
	}

	# Have the steward audit the accounts
	option = {
		name = ie_theft.0001.b
		trigger = { exists = scope:steward }
		ie_theft_set_method_effect = { METHOD = 2 }
		random_list = {
			40 = {
				modifier = {
					add = 15
					scope:steward = { stewardship >= 10 }
				}
				modifier = {
					add = 20
					scope:steward = { stewardship >= 16 }
				}
				trigger_event = {
					id = ie_theft.0002
					days = 3
				}
			}
			45 = {
				trigger_event = {
					id = ie_theft.0003
					days = 3
				}
			}
		}
	}

	# Put the spymaster on it
	option = {
		name = ie_theft.0001.c
		trigger = { exists = scope:spymaster }
		ie_theft_set_method_effect = { METHOD = 3 }
		random_list = {
			40 = {
				modifier = {
					add = 15
					scope:spymaster = { intrigue >= 10 }
				}
				modifier = {
					add = 20
					scope:spymaster = { intrigue >= 16 }
				}
				trigger_event = {
					id = ie_theft.0002
					days = 5
				}
			}
			45 = {
				trigger_event = {
					id = ie_theft.0003
					days = 5
				}
			}
		}
	}

	# Delegation: let the steward handle it quietly
	option = {
		name = ie_theft.0001.d
		trigger = { exists = scope:steward }
		random_list = {
			45 = {
				modifier = {
					add = 15
					scope:steward = { stewardship >= 10 }
				}
				modifier = {
					add = 15
					scope:steward = { stewardship >= 16 }
				}
				send_interface_toast = {
					type = event_toast_text_and_effect_good
					title = ie_theft.0001.d.success.t
					desc = ie_theft.0001.d.success.desc
					left_icon = scope:steward
					right_icon = scope:thief
					add_gold = {
						value = ie_theft_loot_value
						multiply = 0.5
					}
					scope:thief = {
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -15
						}
					}
					ie_theft_end_story_effect = yes
				}
			}
			55 = {
				send_interface_toast = {
					type = event_toast_text_and_effect_bad
					title = ie_theft.0001.d.fail.t
					desc = ie_theft.0001.d.fail.desc
					left_icon = scope:steward
					ie_theft_thief_escapes_effect = yes
				}
			}
		}
	}

	# Accept the loss and move on
	option = {
		name = ie_theft.0001.f
		add_character_modifier = {
			modifier = ie_lax_treasury_modifier
			years = 3
		}
		ie_theft_end_story_effect = yes
	}
}

# 0002 The Culprit Revealed
#####################################
ie_theft.0002 = {
	type = character_event
	title = ie_theft.0002.t
	desc = {
		desc = ie_theft.0002.desc.opening
		first_valid = {
			triggered_desc = {
				trigger = { scope:story = { var:method = 1 } }
				desc = ie_theft.0002.desc.method_1
			}
			triggered_desc = {
				trigger = { scope:story = { var:method = 2 } }
				desc = ie_theft.0002.desc.method_2
			}
			triggered_desc = {
				trigger = { scope:story = { var:method = 3 } }
				desc = ie_theft.0002.desc.method_3
			}
			triggered_desc = {
				trigger = { scope:story = { var:method = 4 } }
				desc = ie_theft.0002.desc.method_4
			}
			triggered_desc = {
				trigger = { scope:story = { var:method = 5 } }
				desc = ie_theft.0002.desc.method_5
			}
		}
		triggered_desc = {
			trigger = { exists = scope:sponsor }
			desc = ie_theft.0002.desc.sponsor
		}
		desc = ie_theft.0002.desc.ending
	}
	theme = dungeon
	left_portrait = {
		character = scope:thief
		animation = fear
	}

	trigger = {
		any_owned_story = { type = ie_story_theft }
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
		# A foreign ruler may turn out to have paid for the theft
		random = {
			chance = 25
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = { is_alive = yes }
				save_scope_as = sponsor
			}
			if = {
				limit = { exists = scope:sponsor }
				scope:story = {
					set_variable = {
						name = harbor_ruler
						value = scope:sponsor
					}
					set_variable = {
						name = thief_sponsored
						value = 1
					}
				}
				scope:sponsor = { save_scope_as = harbor_ruler }
			}
		}
	}

	# Execute
	option = {
		name = ie_theft.0002.a
		ie_theft_fate_execute_effect = { RECOVER = 0.5 }
		stress_impact = {
			compassionate = major_stress_impact_gain
			forgiving = minor_stress_impact_gain
			sadistic = minor_stress_impact_loss
		}
	}

	# Dungeon
	option = {
		name = ie_theft.0002.b
		ie_theft_fate_imprison_effect = { RECOVER = 0.5 }
	}

	# Work off the debt
	option = {
		name = ie_theft.0002.c
		ie_theft_fate_debt_effect = { RECOVER = 0.75 }
	}

	# Exile
	option = {
		name = ie_theft.0002.d
		ie_theft_fate_exile_effect = { RECOVER = 0.25 }
	}

	# Mercy
	option = {
		name = ie_theft.0002.e
		ie_theft_fate_mercy_effect = yes
		stress_impact = {
			compassionate = minor_stress_impact_loss
			forgiving = minor_stress_impact_loss
			vengeful = minor_stress_impact_gain
		}
	}

	# Cover it up for a friend
	option = {
		name = ie_theft.0002.f
		trigger = {
			OR = {
				has_relation_friend = scope:thief
				has_relation_lover = scope:thief
				opinion = {
					target = scope:thief
					value >= 40
				}
			}
		}
		add_hook = {
			target = scope:thief
			type = favor_hook
		}
		scope:thief = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 25
			}
		}
		random_list = {
			70 = {
				trigger_event = {
					id = ie_theft.0010
					years = { 3 5 }
				}
			}
			30 = {
				ie_theft_end_story_effect = yes
			}
		}
	}

	# Hold the thief as proof and confront the foreign patron
	option = {
		name = ie_theft.0002.g
		trigger = { exists = scope:sponsor }
		add_gold = {
			value = ie_theft_loot_value
			multiply = 0.5
		}
		imprison = {
			target = scope:thief
			type = dungeon
		}
		trigger_event = {
			id = ie_theft.0007
			days = 5
		}
	}
}

ie_theft.0003 = {
	type = character_event
	title = ie_theft.0003.t
	desc = ie_theft.0003.desc
	theme = intrigue
	left_portrait = {
		character = scope:scapegoat
		animation = worry
	}

	trigger = {
		any_owned_story = { type = ie_story_theft }
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
	}

	# Punish the accused
	option = {
		name = ie_theft.0003.a
		scope:scapegoat = {
			add_opinion = {
				target = root
				modifier = insulted_opinion
				opinion = -40
			}
		}
		imprison = {
			target = scope:scapegoat
			type = dungeon
		}
		add_gold = {
			value = ie_theft_loot_value
			multiply = 0.25
		}
		scope:story = {
			set_variable = {
				name = scapegoat_punished
				value = 1
			}
		}
		ie_theft_thief_escapes_effect = yes
		trigger_event = {
			id = ie_theft.0005
			months = { 6 12 }
		}
		stress_impact = {
			honest = major_stress_impact_gain
			just = major_stress_impact_gain
		}
	}

	# Something is off - keep digging
	option = {
		name = ie_theft.0003.b
		add_gold = {
			value = ie_theft_loot_value
			multiply = -0.1
		}
		trigger_event = {
			id = ie_theft.0004
			months = { 1 2 }
		}
	}

	# Speak to the accused in private
	option = {
		name = ie_theft.0003.e
		trigger = { diplomacy >= 10 }
		random_list = {
			40 = {
				modifier = {
					add = 15
					diplomacy >= 14
				}
				modifier = {
					add = 15
					scope:scapegoat = {
						opinion = {
							target = root
							value >= 20
						}
					}
				}
				ie_theft_set_method_effect = { METHOD = 4 }
				trigger_event = {
					id = ie_theft.0002
					days = 4
				}
			}
			40 = {
				scope:scapegoat = {
					add_opinion = {
						target = root
						modifier = insult_opinion
						opinion = -15
					}
				}
				ie_theft_thief_escapes_effect = yes
			}
		}
	}

	# Write it off
	option = {
		name = ie_theft.0003.c
		add_character_modifier = {
			modifier = ie_lax_treasury_modifier
			years = 3
		}
		ie_theft_thief_escapes_effect = yes
	}
}

# 0006 The Thief Surfaces
#####################################
ie_theft.0006 = {
	type = character_event
	title = ie_theft.0006.t
	desc = ie_theft.0006.desc
	theme = diplomacy

	trigger = {
		any_owned_story = {
			type = ie_story_theft
			exists = var:thief_at_large
			var:thief = { is_alive = yes }
		}
		any_neighboring_and_across_water_top_liege_realm_owner = {
			is_alive = yes
		}
	}

	immediate = {
		ie_theft_load_scopes_effect = yes
		if = {
			limit = { NOT = { exists = scope:harbor_ruler } }
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = { is_alive = yes }
				save_scope_as = harbor_ruler
			}
			if = {
				limit = { exists = scope:harbor_ruler }
				scope:story = {
					set_variable = {
						name = harbor_ruler
						value = scope:harbor_ruler
					}
				}
			}
		}
		if = {
			limit = { exists = scope:harbor_ruler }
			scope:harbor_ruler = { add_courtier = scope:thief }
		}
	}

	# Have the spymaster snatch him
	option = {
		name = ie_theft.0006.a
		trigger = { exists = scope:spymaster }
		random_list = {
			40 = {
				modifier = {
					add = 15
					scope:spymaster = { intrigue >= 12 }
				}
				modifier = {
					add = 20
					scope:spymaster = { intrigue >= 18 }
				}
				ie_theft_set_method_effect = { METHOD = 6 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			45 = {
				scope:harbor_ruler = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
						opinion = -30
					}
				}
				ie_theft_unrecovered_effect = yes
			}
		}
	}

	# Demand extradition
	option = {
		name = ie_theft.0006.b
		random_list = {
			35 = {
				modifier = {
					add = 15
					diplomacy >= 12
				}
				modifier = {
					add = 25
					scope:harbor_ruler = {
						opinion = {
							target = root
							value >= 20
						}
					}
				}
				ie_theft_set_method_effect = { METHOD = 7 }
				trigger_event = {
					id = ie_theft.0008
					days = 6
				}
			}
			55 = {
				trigger_event = {
					id = ie_theft.0007
					days = 3
				}
			}
		}
	}

	# Offer a bounty
	option = {
		name = ie_theft.0006.d
		add_gold = {
			value = ie_theft_loot_value
			multiply = -0.4
		}
		random_list = {
			55 = {
				ie_theft_set_method_effect = { METHOD = 9 }
				trigger_event = {
					id = ie_theft.0008
					months = { 3 8 }
				}
			}
			45 = {
				ie_theft_unrecovered_effect = yes
			}
		}
	}

	# Let him go
	option = {
		name = ie_theft.0006.c
		ie_theft_unrecovered_effect = yes
		stress_impact = {
			vengeful = minor_stress_impact_gain
			wrathful = minor_stress_impact_gain
		}
	}
}

			add_county_modifier = {

			}

			value = ie_theft_loot_value
			multiply = -0.5
		}
		scope:haven_county = {
			remove_county_modifier = ie_bandit_haven_modifier
		}
		add_prestige = 25
		ie_theft_end_story_effect = yes
	}

	# Lead the hunt yourself
	option = {
		name = ie_theft.0009.d
		random_list = {
			50 = {
				modifier = {
					add = 15
					martial >= 10
				}
				modifier = {
					add = 20
					martial >= 16
				}
				scope:haven_county = {
					remove_county_modifier = ie_bandit_haven_modifier
				}
				add_prestige = 75
				ie_theft_end_story_effect = yes
			}
			40 = {
				add_gold = {
					value = ie_theft_loot_value
					multiply = -0.25
				}
				ie_theft_end_story_effect = yes
			}
		}
	}

	# Buy the bandits off

		name = ie_theft.0009.b

			value = ie_theft_loot_value
			multiply = -0.25
		}

			remove_county_modifier = ie_bandit_haven_modifier
			add_county_modifier = {
				modifier = ie_bandit_haven_modifier
				years = 2
			}
		}
		ie_theft_end_story_effect = yes
	}

	# Do nothing
	option = {
		name = ie_theft.0009.c
		ie_theft_end_story_effect = yes
	}

				}
			}
