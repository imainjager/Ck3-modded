eir_war_won_attacker_v8 = {
	effect = {
		scope:attacker = {
			if = {
				limit = { has_variable = eir_bh_raider }
				scope:attacker = { save_scope_as = eir_bh_lord }
				scope:defender = {
					if = {
						limit = { is_ai = no }
						eir_bh_wave_lost_effect = yes
					}
					else_if = {
						limit = {
							exists = liege
							liege = { is_ai = no }
						}
						liege = { eir_bh_wave_lost_effect = yes }
					}
				}
				remove_variable = eir_bh_raider
				remove_variable = eir_viking_target
			}
			else_if = {
				limit = { has_variable = eir_inv_kind }
				scope:attacker = { save_scope_as = eir_inv_lord }
				scope:attacker.var:eir_viking_target ?= { save_scope_as = eir_target_county }
				scope:defender = {
					if = {
						limit = {
							is_ai = yes
							exists = liege
							liege = { is_ai = no }
						}
						liege = { save_scope_as = eir_inv_player }
					}
					else_if = {
						limit = { is_ai = no }
						save_scope_as = eir_inv_player
					}
				}
				if = {
					limit = { exists = scope:eir_inv_player }
					scope:eir_inv_player = {
						if = {
							limit = { scope:eir_inv_lord.var:eir_inv_kind = 1 }
							trigger_event = eir.0423
						}
						else_if = {
							limit = { scope:eir_inv_lord.var:eir_inv_kind = 2 }
							trigger_event = eir.0425
						}
						else_if = {
							limit = { scope:eir_inv_lord.var:eir_inv_kind = 3 }
							trigger_event = eir.0427
						}
						else = {
							trigger_event = eir.0429
						}
					}
				}
				remove_variable = eir_inv_kind
				remove_variable = eir_viking_target
			}
		}
	}
}

# The Black Host: a 12 percent chance a year from 868, certain from 880, once per game.
eir_blackhost_pulse = {
	trigger = {
		eir_is_gael_ruler_trigger = yes
		is_ai = no
		current_year >= 868
		current_year < 880
		NOT = { has_global_variable = eir_bh_started }
		eir_has_coast_trigger = yes
	}
	random_events = {
		chance_to_happen = 12
		1 = eir.0400
	}
}
