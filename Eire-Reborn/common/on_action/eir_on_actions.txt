# THE CORE RULE: an Irish ruler's top title (duchy and above) is destroyed when they die.
on_death = {
	on_actions = { eir_on_death }
}

eir_on_death = {
	effect = {
		if = {
			limit = {
				eir_is_irish_ruler_trigger = yes
				eir_collapse_active_trigger = yes
				exists = primary_title
				primary_title = { tier >= tier_duchy }
			}
			eir_collapse_top_title_effect = yes
		}
	}
}

# Invasions: report the result to the defender.
on_war_won_attacker = {
	on_actions = { eir_war_won_attacker }
}

eir_war_won_attacker = {
	effect = {
		scope:attacker = {
			if = {
				limit = { has_variable = eir_viking_raider }
				scope:defender = {
					if = {
						limit = { is_ai = no }
						scope:attacker = { save_scope_as = eir_norse_lord }
						scope:attacker.var:eir_viking_target ?= { save_scope_as = eir_target_county }
						trigger_event = eir.0016
					}
				}
				remove_variable = eir_viking_raider
				remove_variable = eir_viking_target
			}
			else_if = {
				limit = { has_variable = eir_foreign_raider }
				scope:defender = {
					if = {
						limit = { is_ai = no }
						scope:attacker = { save_scope_as = eir_foreign_lord }
						scope:attacker.var:eir_viking_target ?= { save_scope_as = eir_target_county }
						trigger_event = eir.0021
					}
				}
				remove_variable = eir_foreign_raider
				remove_variable = eir_viking_target
			}
		}
	}
}
